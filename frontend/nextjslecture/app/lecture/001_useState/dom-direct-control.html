JavaScript로 DOM 직접 조작하기

왼쪽 소스에서 DOM 요소를 찾고, 오른쪽 버튼을 눌러 화면이 바뀌는 것을 확인하세요.

소스 코드

<!-- JavaScript가 찾을 HTML 요소 -->
<p id="favorite-status">즐겨찾기에 저장되지 않았습니다.</p>
<button id="favorite-button" type="button">즐겨찾기 추가</button>

const statusElement = document.getElementById('favorite-status');
const favoriteButton = document.getElementById('favorite-button');
let isFavorite = false;

favoriteButton.addEventListener('click', function () {
    isFavorite = !isFavorite;
    statusElement.textContent = isFavorite
        ? '즐겨찾기에 저장되었습니다.'
        : '즐겨찾기에 저장되지 않았습니다.';
    favoriteButton.textContent = isFavorite
        ? '즐겨찾기 해제'
        : '즐겨찾기 추가';
});

실제 실행 화면
